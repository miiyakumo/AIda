pub fn default_project_dir(name: &str) -> PathBuf {
    let home = std::env::var("HOME").unwrap_or_else(|_| ".".into());
    PathBuf::from(home)
        .join(".alda-agent")
        .join("projects")
        .join(name)
}

pub fn list_projects() -> Result<Vec<(String, PathBuf)>> {
    let home = std::env::var("HOME").unwrap_or_else(|_| ".".into());
    let base = PathBuf::from(home).join(".alda-agent").join("projects");

    if !base.exists() {
        return Ok(Vec::new());
    }

    let mut projects = Vec::new();
    for entry in fs::read_dir(&base).context("无法读取项目目录")? {
        let entry = entry?;
        let path = entry.path();
        if path.is_dir() && path.join("project.json").exists() {
            let name = path
                .file_name()
                .unwrap_or_default()
                .to_string_lossy()
                .to_string();
            projects.push((name, path));
        }
    }
    projects.sort_by(|a, b| a.0.cmp(&b.0));
    Ok(projects)
}

impl Project {
    pub fn load_or_create(path: PathBuf, project_name: &str, source_material: &str) -> Result<Self> {
        let project_file = path.join("project.json");

        if project_file.exists() {
            let content = fs::read_to_string(&project_file)
                .context("无法读取 project.json")?;
            let project: Project = serde_json::from_str(&content)
                .context("无法解析 project.json")?;
            Ok(project)
        } else {
            // 创建新项目目录
            fs::create_dir_all(path.join("versions"))
                .context("无法创建 versions 目录")?;
            fs::create_dir_all(path.join("exports"))
                .context("无法创建 exports 目录")?;

            let project = Project {
                project_name: project_name.to_string(),
                source_material: source_material.to_string(),
                interpretation: String::new(),
                current_version: 0,
                mode: "full".to_string(),
                versions: BTreeMap::new(),
            };

            let json = serde_json::to_string_pretty(&project)
                .context("无法序列化 project.json")?;
            fs::write(&project_file, json)
                .context("无法写入 project.json")?;

            Ok(project)
        }
    }

    fn write_meta(&self, path: &Path) -> Result<()> {
        let project_file = path.join("project.json");
        let json = serde_json::to_string_pretty(self)
            .context("无法序列化 project.json")?;
        fs::write(&project_file, json)
            .context("无法写入 project.json")?;
        Ok(())
    }

    pub fn save_version(
        &mut self,
        path: &Path,
        alda_code: &str,
        summary: &str,
        checks_passed: bool,
    ) -> Result<u32> {
        let next = self.versions.keys().next_back().copied().unwrap_or(0) + 1;

        // 写入版本文件
        let version_file = path
            .join("versions")
            .join(format!("{:04}.alda", next));
        fs::write(&version_file, alda_code)
            .context("无法写入版本文件")?;

        // 更新 current.alda
        let current_file = path.join("current.alda");
        fs::write(&current_file, alda_code)
            .context("无法写入 current.alda")?;

        // 更新元数据
        self.current_version = next;
        self.versions.insert(
            next,
            VersionMeta {
                created_at: chrono_now(),
                summary: summary.to_string(),
                checks_passed,
            },
        );

        self.write_meta(path)?;
        Ok(next)
    }

    pub fn restore_version(&mut self, path: &Path, version: u32) -> Result<()> {
        if !self.versions.contains_key(&version) {
            bail!("版本 {} 不存在", version);
        }

        let version_file = path
            .join("versions")
            .join(format!("{:04}.alda", version));
        let code = fs::read_to_string(&version_file)
            .context("无法读取版本文件")?;

        let current_file = path.join("current.alda");
        fs::write(&current_file, &code)
            .context("无法写入 current.alda")?;

        self.current_version = version;
        self.write_meta(path)?;
        Ok(())
    }

    pub fn current_code(&self, path: &Path) -> Result<String> {
        let current_file = path.join("current.alda");
        if current_file.exists() {
            fs::read_to_string(&current_file).context("无法读取 current.alda")
        } else {
            bail!("尚无当前作品")
        }
    }
}

/// 获取当前 UTC 时间的 ISO 8601 字符串（不带 chrono 依赖）
fn chrono_now() -> String {
    use std::time::SystemTime;
    let dur = SystemTime::now()
        .duration_since(SystemTime::UNIX_EPOCH)
        .unwrap_or_default();
    let secs = dur.as_secs();
    // 简单格式化：YYYY-MM-DDTHH:MM:SSZ
    let days = secs / 86400;
    let time_secs = secs % 86400;
    let hours = time_secs / 3600;
    let minutes = (time_secs % 3600) / 60;
    let seconds = time_secs % 60;

    // 从 epoch 天数计算年月日（简化版）
    let (year, month, day) = days_to_date(days as i64);

    format!(
        "{:04}-{:02}-{:02}T{:02}:{:02}:{:02}Z",
        year, month, day, hours, minutes, seconds
    )
}

fn days_to_date(mut days: i64) -> (i64, u32, u32) {
    // 从 1970-01-01 开始计算
    days += 719468; // 到 0000-03-01 的天数偏移（简化的 Gregorian）
    let era = if days >= 0 { days } else { days - 146096 } / 146097;
    let doe = days - era * 146097;
    let yoe = (doe - doe / 1460 + doe / 36524 - doe / 146096) / 365;
    let y = yoe + era * 400;
    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
    let mp = (5 * doy + 2) / 153;
    let d = doy - (153 * mp + 2) / 5 + 1;
    let m = if mp < 10 { mp + 3 } else { mp - 9 };
    let y = if m <= 2 { y + 1 } else { y };
    (y, m as u32, d as u32)
}

/// 基于用户自然语言反馈修改当前作品
pub async fn modify(&self, request: ModifyRequest) -> Result<CreationResult> {
    let max_rounds = request.max_rounds.max(1);

    let mut messages: Vec<Message> = Vec::new();

    // system
    messages.push(Message {
        role: "system".to_string(),
        content: Some(SYSTEM_PROMPT.to_string()),
        tool_calls: None,
        tool_call_id: None,
    });

    // user: 从稳定项目事实重建干净修改请求，不回灌旧生成历史
    let user_content = build_modify_user_message(&request);
    messages.push(Message {
        role: "user".to_string(),
        content: Some(user_content),
        tool_calls: None,
        tool_call_id: None,
    });

    // 与 create() 相同的修正循环
    let mut interpretation = String::new();
    let mut last_alda_code: Option<String> = None;
    let mut last_checks: Vec<AldaCheck> = Vec::new();
    let mut was_truncated = false;

    for round in 0..max_rounds {
        let tools = vec![submit_alda_tool()];

        let events = self
            .client
            .chat_stream(messages.clone(), Some(tools))
            .await?;

        let mut tool_call_args: Option<(Option<String>, String, String)> = None;

        for event in &events {
            match event {
                StreamEvent::Text(text) => {
                    interpretation.push_str(text);
                }
                StreamEvent::ToolCall {
                    id,
                    name,
                    arguments,
                } => {
                    if name == "submit_alda" {
                        tool_call_args = Some((id.clone(), name.clone(), arguments.clone()));
                    }
                }
                StreamEvent::Done { finish_reason } => {
                    if finish_reason == "length" {
                        was_truncated = true;
                    }
                }
            }
        }

        let (tool_id, _, tool_args) = match tool_call_args {
            Some(args) => args,
            None => bail!("模型未提交 Alda 乐谱（未调用 submit_alda 工具）"),
        };

        let alda_code = parse_alda_code_from_args(&tool_args)?;

        let tmp_dir = tempfile::tempdir().context("创建临时目录失败")?;
        let tmp_score = tmp_dir.path().join("candidate.alda");
        fs::write(&tmp_score, &alda_code)?;

        let mut checks = self.runner.validate(
            &tmp_score,
            &request.included_instruments,
            &request.excluded_instruments,
            None,  // modify 不检查时长
            10.0,
        );

        if was_truncated {
            checks.push(AldaCheck {
                name: "输出完整性",
                status: CheckStatus::Fail,
                detail: "模型输出被截断（达到 token 限制），作品可能不完整".to_string(),
            });
        }

        let all_pass = checks.iter().all(|c| c.status != CheckStatus::Fail);

        last_alda_code = Some(alda_code.clone());
        last_checks = checks.clone();

        if all_pass {
            return Ok(CreationResult {
                rounds: round + 1,
                success: true,
                checks,
                alda_code: Some(alda_code),
                interpretation,
                was_truncated,
            });
        }

        let feedback = build_tool_feedback(&checks, tool_id.as_deref());
        let tc_id = tool_id.clone().unwrap_or_else(|| "call_1".to_string());

        messages.push(Message {
            role: "assistant".to_string(),
            content: None,
            tool_calls: Some(vec![crate::deepseek::ToolCallMsg {
                id: tc_id,
                ty: "function".to_string(),
                function: crate::deepseek::FunctionCallArgs {
                    name: "submit_alda".to_string(),
                    arguments: tool_args,
                },
            }]),
            tool_call_id: None,
        });

        messages.push(Message {
            role: "tool".to_string(),
            content: Some(feedback),
            tool_calls: None,
            tool_call_id: tool_id,
        });
    }

    Ok(CreationResult {
        rounds: max_rounds,
        success: false,
        checks: last_checks,
        alda_code: last_alda_code,
        interpretation,
        was_truncated,
    })
}
```

    NaturalLanguage(String),  // 自然语言输入
}

fn parse_input(input: &str) -> ReplAction {
    let trimmed = input.trim();
    if trimmed.starts_with('/') {
        ReplAction::Command(trimmed.to_string())
    } else if !trimmed.is_empty() {
        ReplAction::NaturalLanguage(trimmed.to_string())
    } else {
        ReplAction::Command(String::new()) // 空行，忽略
    }
}

pub async fn run_repl(project_dir: PathBuf) -> Result<()> {
    // 初始化基础设施
    let config = Config::from_env_file()
        .context("无法读取配置，请确保 ../.env 文件存在")?;

    let client = DeepSeekClient::new(
        config.api_key.clone(),
        config.base_url.clone(),
        config.model.clone(),
    )?;

    let alda_path = find_alda()
        .ok_or_else(|| anyhow::anyhow!("未找到 alda，请先安装"))?;
    let runner = AldaRunner::new(alda_path);
    let agent = Agent::new(client, runner.clone());

    // 加载或创建项目
    let project_name = project_dir
        .file_name()
        .map(|n| n.to_string_lossy().to_string())
        .unwrap_or_else(|| "untitled".to_string());
    let mut project = Project::load_or_create(
        project_dir.clone(),
        &project_name,
        "",
    )?;

    println!("\nAlda Music Agent — {}", project_name);
    println!("输入 /help 查看命令，自然语言输入即可创作或修改。\n");

    // 只在当前自动修正或澄清往返中暂存消息，完成后清空
    let mut conversation: Vec<Message> = Vec::new();

    loop {
        print!("> ");
        io::stdout().flush().ok();

        let mut input = String::new();
        if io::stdin().read_line(&mut input).is_err() {
            break;
        }

        match parse_input(&input) {
            ReplAction::Command(cmd) => {
                let parts: Vec<&str> = cmd.split_whitespace().collect();
                match parts[0] {
                    "/help" => println!("{}", HELP_TEXT),
                    "/quit" | "/exit" => {
                        println!("再见。");
                        break;
                    }
                    "/play" => {
                        let current_file = project_dir.join("current.alda");
                        if current_file.exists() {
                            match runner.play(&current_file) {
                                Ok(()) => println!("▶  开始播放"),
                                Err(e) => println!("播放失败: {}", e),
                            }
                        } else {
                            println!("尚无作品可播放。");
                        }
                    }
                    "/stop" => {
                        match runner.stop() {
                            Ok(()) => println!("■ 停止播放"),
                            Err(e) => println!("停止失败: {}", e),
                        }
                    }
                    "/export" => {
                        let current_file = project_dir.join("current.alda");
                        if current_file.exists() {
                            let export_dir = project_dir.join("exports");
                            fs::create_dir_all(&export_dir).ok();

                            // 复制 .alda
                            let v = project.current_version;
                            let alda_export = export_dir
                                .join(format!("v{:04}.alda", v));
                            fs::copy(&current_file, &alda_export)
                                .context("导出 .alda 失败")?;
                            println!("✅ 导出 .alda: {}", alda_export.display());

                            // 导出 MIDI
                            let midi_export = export_dir
                                .join(format!("v{:04}.mid", v));
                            match runner.export_midi(&current_file, &midi_export) {
                                Ok(p) => println!("✅ 导出 MIDI: {}", p.display()),
                                Err(e) => println!("MIDI 导出失败: {}", e),
                            }
                        } else {
                            println!("尚无作品可导出。");
                        }
                    }
                    "/history" => {
                        if project.versions.is_empty() {
                            println!("尚无版本记录。");
                        } else {
                            println!("版本历史:");
                            for (n, meta) in project.versions.iter() {
                                let current = if *n == project.current_version {
                                    " ← 当前"
                                } else {
                                    ""
                                };
                                let status = if meta.checks_passed { "✅" } else { "❌" };
                                println!(
                                    "  {}  v{} {} {} — {}{}",
                                    status, n, meta.created_at, meta.summary, meta.summary, current
                                );
                            }
                        }
                    }
                    "/restore" => {
                        if parts.len() < 2 {
                            println!("用法: /restore <版本号>");
                        } else {
                            match parts[1].parse::<u32>() {
                                Ok(n) => match project.restore_version(&project_dir, n) {
                                    Ok(()) => {
                                        println!("已恢复版本 {}。", n);
                                        // 清空短期澄清/自动修正上下文，因为状态已改变
                                        conversation.clear();
                                    }
                                    Err(e) => println!("恢复失败: {}", e),
                                },
                                Err(_) => println!("无效的版本号。"),
                            }
                        }
                    }
                    "" => {} // 空行忽略
                    other => println!("未知命令: {}. 输入 /help 查看可用命令。", other),
                }
            }
            ReplAction::NaturalLanguage(text) => {
                let has_current = project_dir.join("current.alda").exists()
                    && project.current_version > 0;

                if has_current {
                    // 修改模式
                    println!("⏳ 处理修改请求...");
                    let current_code = project
                        .current_code(&project_dir)
                        .unwrap_or_default();

                    let modify_req = ModifyRequest {
                        source_material: project.source_material.clone(),
                        current_alda: current_code,
                        feedback: text,
                        mode: CreationMode::FullPiece,
                        target_duration_secs: None,
                        included_instruments: Vec::new(),
                        excluded_instruments: Vec::new(),
                        max_rounds: 3,
                    };

                    match agent.modify(modify_req).await {
                        Ok(result) => {
                            println!();
                            if result.success {
                                if let Some(ref code) = result.alda_code {
                                    let v = project.save_version(
                                        &project_dir,
                                        code,
                                        "用户修改",
                                        true,
                                    )?;
                                    println!("✅ 修改成功 — 版本 {}", v);
                                }
                            } else {
                                println!("❌ 修改未通过检查:");
                                for check in &result.checks {
                                    let icon = match check.status {
                                        crate::alda::CheckStatus::Pass => "✅",
                                        crate::alda::CheckStatus::Fail => "❌",
                                        crate::alda::CheckStatus::Unchecked => "⏭",
                                    };
                                    println!("  {} {}: {}", icon, check.name, check.detail);
                                }
                                println!("\n当前作品未受影响。请尝试更具体的修改指令。");
                            }
                        }
                        Err(e) => {
                            println!("修改失败: {}", e);
                        }
                    }
                } else {
                    // 创作模式
                    println!("⏳ 开始创作...");
                    let create_req = CreationRequest {
                        source_material: text.clone(),
                        mode: CreationMode::FullPiece,
                        target_duration_secs: None,
                        included_instruments: Vec::new(),
                        excluded_instruments: Vec::new(),
                        max_rounds: 3,
                    };

                    match agent.create(create_req).await {
                        Ok(result) => {
                            println!();
                            if result.success {
                                if let Some(ref code) = result.alda_code {
                                    // 更新项目元数据
                                    project.interpretation = result.interpretation.clone();
                                    project.source_material = text;

                                    let v = project.save_version(
                                        &project_dir,
                                        code,
                                        "首次创作",
                                        true,
                                    )?;
                                    println!("✅ 创作完成 — 版本 {}", v);

                                    // 显示检查结果
                                    for check in &result.checks {
                                        let icon = match check.status {
                                            crate::alda::CheckStatus::Pass => "✅",
                                            crate::alda::CheckStatus::Fail => "❌",
                                            crate::alda::CheckStatus::Unchecked => "⏭",
                                        };
                                        println!("  {} {}: {}", icon, check.name, check.detail);
                                    }

                                    if !result.interpretation.is_empty() {
                                        println!("\n--- 音乐构想 ---");
                                        println!("{}", result.interpretation);
                                    }
                                }
                            } else {
                                println!("❌ 创作未通过检查:");
                                for check in &result.checks {
                                    let icon = match check.status {
                                        crate::alda::CheckStatus::Pass => "✅",
                                        crate::alda::CheckStatus::Fail => "❌",
                                        crate::alda::CheckStatus::Unchecked => "⏭",
                                    };
                                    println!("  {} {}: {}", icon, check.name, check.detail);
                                }
                                if result.was_truncated {
                                    println!("\n⚠️  模型输出被截断，作品可能不完整。");
                                }
                            }
                        }
                        Err(e) => {
                            println!("创作失败: {}", e);
                        }
                    }
                }
            }
        }
    }

    Ok(())
}
```

注意：`repl.rs` 中需要 `use std::fs;` 用于 `/export` 命令。

async fn main() -> anyhow::Result<()> {

    match cli.command {

            let projects = alda_agent::project::list_projects()?;
            if projects.is_empty() {
                println!("尚无项目。在 REPL 中创建新项目或使用 --project 指定目录。");

            mode,
            duration,
            include,
            exclude,
            output,
        }) => create(file, mode, duration, include, exclude, output).await,
        None => {
            // 进入 REPL
            let project_dir = if let Some(dir) = cli.project {
                dir
            } else {
                // 列出项目，让用户选择或创建
                let projects = alda_agent::project::list_projects()?;
                if projects.is_empty() {
                    eprintln!("尚无项目。请指定项目路径: --project <路径>");
                    eprintln!("或使用一次性创作: alda-agent create ...");
                    return Ok(());
                }
                if projects.len() == 1 {
                    println!("进入项目: {}", projects[0].0);
                    projects[0].1.clone()
                } else {
                    println!("已有项目:");
                    for (i, (name, path)) in projects.iter().enumerate() {
                        println!("  [{}] {}  ({})", i + 1, name, path.display());
                    }
                    println!("\n请选择项目编号 (1-{}) 或输入新项目名称:", projects.len());

                    use std::io::{self, Write};
                    print!("> ");
                    io::stdout().flush().ok();

                    let mut choice = String::new();
                    io::stdin().read_line(&mut choice)?;
                    let choice = choice.trim();

                    if let Ok(n) = choice.parse::<usize>()
                        && n >= 1 && n <= projects.len()
                    {
                        projects[n - 1].1.clone()
                    } else if !choice.is_empty() {
                        // 创建新项目
                        let new_dir =
                            alda_agent::project::default_project_dir(choice);
                        println!("创建新项目: {} ({})", choice, new_dir.display());
                        new_dir
                    } else {
                        anyhow::bail!("未选择项目");
                    }
                }
            };

            alda_agent::repl::run_repl(project_dir).await?;
        }
    }

    Ok(())
}
